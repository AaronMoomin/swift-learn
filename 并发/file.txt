    /**
     * 根据错误题目数量计算正确率
     * @param {number} errorCount - 错误题目数量
     * @param {number} totalQuestions - 总题目数量
     * @returns {string} 正确率等级（"80%及以上"、"50%-70%"、"40%及以下"）
     */
    calculateAccuracyLevel: function(errorCount, totalQuestions) {
        // 输入验证
        if (typeof errorCount !== 'number' || typeof totalQuestions !== 'number') {
            throw new Error("错误：错误题目数和总题目数必须为数字类型");
        }
        if (errorCount < 0 || totalQuestions <= 0 || errorCount > totalQuestions) {
            throw new Error("错误：错误题目数不能为负数，且不能超过总题目数；总题目数必须大于0");
        }

        const correctCount = totalQuestions - errorCount;
        const accuracy = (correctCount / totalQuestions) * 100;

        if (accuracy >= 80) {
            return "80%及以上";
        } else if (accuracy >= 50 && accuracy < 80) {
            return "50%-70%";
        } else {
            return "40%及以下";
        }
    },

    /**
     * 根据各题目区块的错误数量判定最终分数
     * @param {number} errors1_10 - 题目1-10的错误数量
     * @param {number} errors11_20 - 题目11-20的错误数量
     * @param {number} errors21_30 - 题目21-30的错误数量
     * @returns {string} 最终判定的分数
     */
    determineScore: function(errors1_10, errors11_20, errors21_30) {
        // 输入验证
        const validateErrorCount = (count, section) => {
            if (typeof count !== 'number' || count < 0 || count > 10) {
                throw new Error(`错误：${section}的错误数量必须为0-10之间的数字`);
            }
        };

        validateErrorCount(errors1_10, "题目1-10");
        validateErrorCount(errors11_20, "题目11-20");
        validateErrorCount(errors21_30, "题目21-30");

        // 第一步：判断题目1-10
        if (errors1_10 >= 3 && errors1_10 <= 5) {
            return "3-3.5分";
        } else if (errors1_10 >= 6) {
            return "接近零基础";
        } else {
            // 错误0-2题，继续判断题目11-20
            if (errors11_20 >= 0 && errors11_20 <= 2) {
                return "5.5分";
            } else if (errors11_20 >= 6) {
                return "4分";
            } else {
                // 错误3-5题，继续判断题目21-30
                if (errors21_30 >= 0 && errors21_30 <= 2) {
                    return "5.5分";
                } else if (errors21_30 >= 3 && errors21_30 <= 5) {
                    return "4.5-5分";
                } else {
                    // 错误6个以上
                    return "4-4.5分";
                }
            }
        }
    },

    /**
     * 生成完整的测评报告
     * @param {number} errors1_10 - 题目1-10的错误数量
     * @param {number} errors11_20 - 题目11-20的错误数量
     * @param {number} errors21_30 - 题目21-30的错误数量
     * @param {Object} sectionConfig - 各部分题目配置（基础语法、进阶语法、词汇的题目分布）
     * @returns {Object} 完整的测评报告
     */
    generateReport: function(errors1_10, errors11_20, errors21_30, sectionConfig = {}) {
        try {
            // 1. 判定最终分数
            const finalScore = this.determineScore(errors1_10, errors11_20, errors21_30);
            
            // 2. 获取综合评估话术
            const comprehensiveEvaluation = this.scoreEvaluationTexts[finalScore] || 
                "暂无对应评估话术，请检查分数判定是否正确";

            // 3. 处理各部分题目配置（默认配置：题目1-10=基础语法，11-20=进阶语法，21-30=词汇）
            const config = {
                基础语法: { start: 1, end: 10, total: 10 },
                进阶语法: { start: 11, end: 20, total: 10 },
                词汇: { start: 21, end: 30, total: 10 },
                ...sectionConfig
            };

            // 4. 计算各部分的错误数量和正确率分析
            const sectionAnalysis = {};
            let totalErrors = errors1_10 + errors11_20 + errors21_30;

            // 处理基础语法部分
            sectionAnalysis.基础语法 = this._getSectionAnalysis(
                errors1_10, 
                config.基础语法.total, 
                "基础语法"
            );

            // 处理进阶语法部分
            sectionAnalysis.进阶语法 = this._getSectionAnalysis(
                errors11_20, 
                config.进阶语法.total, 
                "进阶语法"
            );

            // 处理词汇部分
            sectionAnalysis.词汇 = this._getSectionAnalysis(
                errors21_30, 
                config.词汇.total, 
                "词汇"
            );

            // 5. 生成完整报告
            return {
                测评结果: {
                    题目1_10错误数: errors1_10,
                    题目11_20错误数: errors11_20,
                    题目21_30错误数: errors21_30,
                    总错误数: totalErrors,
                    预估分数: finalScore
                },
                综合评估: comprehensiveEvaluation,
                各部分详细分析: sectionAnalysis,
                生成时间: new Date().toLocaleString()
            };
        } catch (error) {
            console.error("生成测评报告失败:", error.message);
            return {
                状态: "失败",
                错误信息: error.message,
                生成时间: new Date().toLocaleString()
            };
        }
    },

    /**
     * 内部方法：获取单个部分的分析结果
     * @param {number} errorCount - 该部分错误数量
     * @param {number} totalQuestions - 该部分总题目数
     * @param {string} sectionName - 部分名称
     * @returns {Object} 该部分的分析结果
     */
    _getSectionAnalysis: function(errorCount, totalQuestions, sectionName) {
        const accuracyLevel = this.calculateAccuracyLevel(errorCount, totalQuestions);
        const correctCount = totalQuestions - errorCount;
        const accuracy = ((correctCount / totalQuestions) * 100).toFixed(1) + "%";

        return {
            错误题目数: errorCount,
            正确题目数: correctCount,
            总题目数: totalQuestions,
            正确率: accuracy,
            正确率等级: accuracyLevel,
            分析话术: this.accuracyAnalysisTexts[sectionName][accuracyLevel] || 
                "暂无对应分析话术，请检查配置是否正确"
        };
    },

            { name: "测试7: 1-10错3题", e1: 3, e2: 0, e3: 0, expected: "3-3.5分" },
            { name: "测试8: 1-10错5题", e1: 5, e2: 10, e3: 10, expected: "3-3.5分" },
            { name: "测试9: 1-10错6题", e1: 6, e2: 0, e3: 0, expected: "接近零基础" },

        ];

        // 执行测试
        testCases.forEach((test, index) => {
            try {
                const result = this.determineScore(test.e1, test.e2, test.e3);
                const status = result === test.expected ? "✅ 成功" : "❌ 失败";
                console.log(`${status} ${test.name}`);
                console.log(`   输入: 题目1-10错${test.e1}题, 题目11-20错${test.e2}题, 题目21-30错${test.e3}题`);
                console.log(`   预期: ${test.expected}, 实际: ${result}`);
                console.log();
            } catch (error) {
                console.log(`❌ 错误 ${test.name}: ${error.message}`);
                console.log();
            }
        });

        console.log("测试执行完成！");
    }
};
